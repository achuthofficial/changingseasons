-- Makes deleted_at authoritative by stamping it server-side. Run after
-- supabase-soft-delete-restore.sql.txt. Idempotent, safe to re-run.
--
-- The app sends `deleted_at: new Date().toISOString()` — the *browser's*
-- clock — while purge_deleted_records() compares against now(), the
-- *database's* clock. A shop PC running half an hour fast had its records
-- purged half an hour early, while the countdown on screen still showed
-- time remaining and Restore failed on a record that looked recoverable.
-- A slow clock kept records past their window instead. Windows machines
-- that sleep a lot or aren't syncing time drift like this routinely.
--
-- This overwrites whatever the client sent with the server's own clock, so
-- the countdown, the purge, and the record all agree no matter what state
-- the browser's clock is in.
--
-- Note on the cascade: now() is transaction_timestamp(), which is constant
-- for the whole transaction. A customer and the orders and payments its
-- triggers cascade to are all updated inside one transaction, so they still
-- receive one identical timestamp. The batch id that restore depends on
-- survives — it just comes from the database now instead of the browser.

create or replace function public.stamp_deleted_at()
returns trigger
language plpgsql
as $$
begin
  -- Only on the transition into deleted. Restores (deleted_at -> null) and
  -- updates that merely carry the column along unchanged are left alone.
  if new.deleted_at is not null and old.deleted_at is null then
    new.deleted_at := now();
  end if;
  return new;
end;
$$;

drop trigger if exists trg_stamp_deleted_at_users on public.users;
create trigger trg_stamp_deleted_at_users
  before update of deleted_at on public.users
  for each row execute function public.stamp_deleted_at();

drop trigger if exists trg_stamp_deleted_at_orders on public.orders;
create trigger trg_stamp_deleted_at_orders
  before update of deleted_at on public.orders
  for each row execute function public.stamp_deleted_at();

drop trigger if exists trg_stamp_deleted_at_staff on public.staff;
create trigger trg_stamp_deleted_at_staff
  before update of deleted_at on public.staff
  for each row execute function public.stamp_deleted_at();

drop trigger if exists trg_stamp_deleted_at_transactions on public.transactions;
create trigger trg_stamp_deleted_at_transactions
  before update of deleted_at on public.transactions
  for each row execute function public.stamp_deleted_at();

drop trigger if exists trg_stamp_deleted_at_expenses on public.expenses;
create trigger trg_stamp_deleted_at_expenses
  before update of deleted_at on public.expenses
  for each row execute function public.stamp_deleted_at();

-- Anything deleted before this file ran still carries a browser timestamp.
-- Pulling any of those that are implausibly far from server time back to
-- now() gives them a full, correct window rather than a wrong one. Rows
-- already legitimately near expiry (within the normal hour) are untouched.
update public.users        set deleted_at = now() where deleted_at > now() or deleted_at < now() - interval '60 minutes';
update public.orders       set deleted_at = now() where deleted_at > now() or deleted_at < now() - interval '60 minutes';
update public.staff        set deleted_at = now() where deleted_at > now() or deleted_at < now() - interval '60 minutes';
update public.transactions set deleted_at = now() where deleted_at > now() or deleted_at < now() - interval '60 minutes';
update public.expenses     set deleted_at = now() where deleted_at > now() or deleted_at < now() - interval '60 minutes';
