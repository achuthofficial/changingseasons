-- Links transactions to the order they were paid against, and backfills a
-- catch-up transaction for existing orders whose advance_paid was set
-- directly on the order form (before payment recording always went through
-- a real transaction). Run after supabase-order-items-schema.sql.txt.

alter table public.transactions add column if not exists order_id bigint references public.orders(id) on delete set null;

-- Backfill: one row per order that has an advance recorded but no linked
-- transaction for it yet. Dated to the order's created_at since we don't
-- know when the money actually changed hands. Method 'Unknown' marks these
-- as backfilled rather than actually recorded through Record Payment.
-- Idempotent: the `not exists` guard means this is safe to re-run and will
-- never double-insert for an order that already has a linked transaction.
insert into public.transactions (order_id, customer, email, item, amount, method, status, created_at)
select
  o.id,
  u.name,
  u.email,
  'Advance payment (recorded before payment tracking)',
  o.advance_paid,
  'Unknown',
  'Completed',
  o.created_at
from public.orders o
join public.users u on u.id = o.customer_id
where o.advance_paid > 0
  and not exists (select 1 from public.transactions t where t.order_id = o.id);
